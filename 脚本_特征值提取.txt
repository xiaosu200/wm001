--下单时间距注册时间间隔
--drop table temp_bi.sumt_anls_eigen_value_01;
create table temp_bi.sumt_anls_eigen_value_01 as
select a.order_id,a.order_sn,a.valid_flag,a.user_id,a.add_time,b.register_time,
       case when b.register_time is null then (unix_timestamp(a.add_time)-unix_timestamp('2013-03-11 00:00:00'))/3600 else (unix_timestamp(a.add_time)-unix_timestamp(b.register_time))/3600 end as add_register_time
from 
(select order_id,order_sn,user_id,add_time,valid_flag
 from temp_bi.sumt_anls_order_list where add_month>=201501 and add_month<=202107
 group by order_id,order_sn,user_id,add_time,valid_flag) a
left join 
(select user_id,register_time
 from temp_bi.sumt_anls_register_list    --脚本_客户养成规律清单
 group by user_id,register_time) b on a.user_id=b.user_id;

 
--下单间隔(相邻订单间隔的分钟数)
--temp_bi.sumt_anls_eigen_value_02_bak(202106)
drop table temp_bi.sumt_anls_eigen_value_02;
create table temp_bi.sumt_anls_eigen_value_02 as
select order_id,add_time,drk_time,add_time_before,add_time_next,
       case when add_time_before is not null then (unix_timestamp(add_time)-unix_timestamp(add_time_before))/3600 else -99 end as add_before_time,
       case when add_time_next is not null then (unix_timestamp(add_time_next)-unix_timestamp(add_time))/3600 else -99 end as add_next_time
from (
  select a.order_id,a.add_time,a.drk_time,b.add_time as add_time_before,c.add_time as add_time_next
         --b.order_id as order_id_before,b.add_time as add_time_before,b.drk_time as drk_time_before,
         --c.order_id as order_id_next,c.add_time as add_time_next,c.drk_time as drk_time_next
  from 
  (select order_id,user_id,add_time,dense_rank() over (partition by user_id order by add_time asc) drk_time 
   from temp_bi.sumt_anls_order_list where add_month>=201501 and add_month<=202107
   group by order_id,user_id,add_time) a
  left join 
  (select order_id,user_id,add_time,dense_rank() over (partition by user_id order by add_time asc) drk_time 
   from temp_bi.sumt_anls_order_list where add_month>=201501 and add_month<=202107
   group by order_id,user_id,add_time ) b
   on a.user_id=b.user_id and a.drk_time=b.drk_time+1 --上一单
  left join 
  (select order_id,user_id,add_time,dense_rank() over (partition by user_id order by add_time asc) drk_time 
   from temp_bi.sumt_anls_order_list where add_month>=201501 and add_month<=202107
   group by order_id,user_id,add_time ) c
   on a.user_id=c.user_id and a.drk_time=c.drk_time-1 --下一单
  group by  a.order_id,a.add_time,a.drk_time,b.add_time,c.add_time
 ) x
 
 
--领券时间距注册时间间隔 ---没有领券的将时间设置为无穷大
drop table temp_bi.sumt_anls_eigen_value_03;
create table temp_bi.sumt_anls_eigen_value_03 as
select a.order_id,a.order_sn,a.user_id,a.add_time,a.register_time,b.activate_time,b.use_time,
       case when b.activate_time is null then 10000 else (unix_timestamp(b.activate_time)-unix_timestamp(a.register_time))/3600 end as activate_register_time
from 
(select a.order_id,a.order_sn,a.user_id,a.add_time,b.register_time
 from (select order_id,order_sn,user_id,add_time from temp_bi.sumt_anls_order_list where add_month>=201501 and add_month<=202107 ) a
 left join 
 (select user_id,register_time
  from temp_bi.sumt_anls_register_list
  group by user_id,register_time) b on a.user_id=b.user_id
 group by a.order_id,a.order_sn,a.user_id,a.add_time,b.register_time) a
left join 
(select use_order_sn,min(activate_time) as activate_time,use_time  ---一个订单会不会对应多张券，取最早的券时间
 from (select user_id,coupon_sn,use_order_sn,
         activate_time,--激活时间	
         use_time,--使用时间	
         begin_time,--开始时间	
         end_time --结束时间
  from dw.dw_pmsapi_coupon_instance 
  where dt=get_dt_date(get_date(-1)) and status=0 --已使用
 	 ) x
 group by use_order_sn,use_time 
) b on a.order_sn=b.use_order_sn;

--=================3.订单特征===================--
--订单优惠力度/商品件数/件单价
--drop table temp_bi.sumt_anls_eigen_value_07;
create table temp_bi.sumt_anls_eigen_value_07 as
select order_id,user_id,add_time,
       case when (quality_order_goods_amt=0 or quality_order_goods_amt is null) then total_goods_amt else quality_order_goods_amt end as order_goods_amt,--订单金额  
       case when (quality_order_goods_amt=0 or quality_order_goods_amt is null) then fav_amt*1.0/total_goods_amt else fav_amt*1.0/quality_order_goods_amt end as quality_order_fav,--优惠力度       
       quality_order_goods_cnt,--合格订单商品件数（含拒退）
       case when total_goods_cnt is null then 0 else total_goods_cnt end as total_goods_cnt,
       case when (quality_order_goods_cnt is null or quality_order_goods_cnt=0) then total_goods_cnt else quality_order_goods_cnt end as order_goods_cnt,
       case when (quality_order_goods_amt=0 or quality_order_goods_amt is null) then total_goods_amt*1.0/total_goods_cnt else quality_order_goods_amt*1.0/quality_order_goods_cnt end as quality_order_goods_price, --商品件单价
       order_reject --是否拒收
from temp_bi.sumt_anls_order_fav_list
where year(add_time)*100+month(add_time)>=201601 and year(add_time)*100+month(add_time)<=202107;

--订单品类是否单品类/品类名称or多品类/品类数量
drop table if exists temp_bi.sumt_anls_eigen_value_08;
create table temp_bi.sumt_anls_eigen_value_08 as
select a.order_id,a.user_id,a.add_time,
       case when a.num=1 then 1 else 0 end as is_single_cate,--是否单品类
       a.num as goods_cate_num, --品类数
	   case when a.num=1 then b.goods_cate else '多品类' end as goods_cate
from
(select order_id,user_id,add_time,count(distinct goods_cate) as num
 from temp_bi.sumt_anls_order_list 
 where add_month>=201501 and add_month<=202107
 group by order_id,user_id,add_time) a
 left join
 (select a2.order_id,a2.goods_cate
  from 
 (select order_id,count(distinct goods_cate)
   from temp_bi.sumt_anls_order_list 
   where  add_month>=201501
   group by order_id
   having count(distinct goods_cate)=1 ) a1 ---单品类订单
   join (select order_id,goods_cate from temp_bi.sumt_anls_order_list 
        where  add_month>=201501 group by order_id,goods_cate) a2 on a1.order_id=a2.order_id ---关联取品类名称
  group by  a2.order_id,a2.goods_cate --去重
  ) b
 on a.order_id=b.order_id 
group by a.order_id,a.user_id,a.add_time,
       case when a.num=1 then 1 else 0 end,--是否单品类
       a.num, --品类数
	   case when a.num=1 then b.goods_cate else '多品类' end;
